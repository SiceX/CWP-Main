#  Mitsubishi
company_mitsubishi = {
    icon = "gfx/interface/icons/generic_icons/unused/lack.dds"
    background = "gfx/interface/icons/company_icons/company_backgrounds/comp_illu_manufacturing_heavy.dds"

    flavored_company = yes

    building_types = {
       building_shipyards
       building_electronics_manufactories
       building_motor_industry
    }

    potential = {
       has_interest_marker_in_region = region_japan
    }

    attainable = {
       hidden_trigger = {
          any_scope_state = {
             state_region = s:STATE_KANTO
          }
       }
    }

    possible = {
       any_scope_state = {
          state_region = s:STATE_KANTO
          is_incorporated = yes
          any_scope_building = {
             is_building_type = building_shipyards
             level >= 5
          }
       }
       has_technology_researched = steam_engines
    }

    prosperity_modifier = {
       building_shipyards_throughput_add = 0.1
       building_motor_industry_throughput_add = 0.1
    }

    ai_weight = {
       value = 3
    }
}
